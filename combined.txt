    Private Sub SetupProfGrid()

            .Columns.Add("colEmpId", "Employee ID")
            .Columns.Add("colDept", "Department")
            .Columns.Add("colClasses", "Assigned Class")
            .Columns.Add("colStatus", "Status")

    Private Sub ApplyProfFilter()
        Dim q As String = txtSearchProf.Text.Trim().ToLower()
        Dim dept As String = If(cboDeptProf.SelectedItem Is Nothing, "All", cboDeptProf.SelectedItem.ToString())

        filteredProfs = allProfs.Where(Function(p) _
            (q = "" OrElse p.Name.ToLower().Contains(q) OrElse p.EmployeeID.ToLower().Contains(q)) AndAlso
            (dept = "All" OrElse p.Department = dept)).ToList()

        profPage = 1
        ShowProfPage()
    End Sub

    Private Sub ShowProfPage()
        dgvProf.Rows.Clear()
        For Each p In filteredProfs.Skip((profPage - 1) * ProfPageSize).Take(ProfPageSize)
            dgvProf.Rows.Add(False, p.Name, p.EmployeeID, p.Department, p.Classes & " classes", p.Status, "")
        Next
        lblTotalProf.Text = "Total Professors: " & filteredProfs.Count
        dgvProf.ClearSelection()
        BuildProfPager()
    End Sub

    ' Always shows at least 5 page buttons; slides a window of 5 when there are more pages
    Private Sub BuildProfPager()
        flowProfPager.SuspendLayout()
        flowProfPager.Controls.Clear()

        Const MinButtons As Integer = 5
        Dim totalPages As Integer = Math.Max(1, CInt(Math.Ceiling(filteredProfs.Count / ProfPageSize)))

        Dim startPage As Integer = Math.Max(1, profPage - 2)
        If totalPages > MinButtons Then
            startPage = Math.Min(startPage, totalPages - MinButtons + 1)
        Else
            startPage = 1
        End If

        ' Previous
        flowProfPager.Controls.Add(MakePagerButton("<", Math.Max(1, profPage - 1), False, profPage > 1))

        ' Number buttons (pages that don't exist are shown disabled)
        For i As Integer = 0 To MinButtons - 1
            Dim pageNo As Integer = startPage + i
            Dim exists As Boolean = pageNo <= totalPages
            flowProfPager.Controls.Add(MakePagerButton(pageNo.ToString(), pageNo, pageNo = profPage, exists))
        Next

        ' Next
        flowProfPager.Controls.Add(MakePagerButton(">", Math.Min(totalPages, profPage + 1), False, profPage < totalPages))

        ' Make sure all buttons fit (fixes the clipped pager)
        flowProfPager.AutoSize = False
        flowProfPager.Width = flowProfPager.Controls.Count * 32 + 8

        flowProfPager.ResumeLayout()
    End Sub

    Private Function MakePagerButton(text As String, page As Integer, active As Boolean, enabled As Boolean) As Button
        Dim b As New Button()
        b.Text = text
        b.Size = New Size(28, 28)
        b.Margin = New Padding(2)
        b.FlatStyle = FlatStyle.Flat
        b.FlatAppearance.BorderColor = If(enabled, Color.FromArgb(0, 100, 0), Color.Silver)
        b.BackColor = If(active, Color.FromArgb(0, 100, 0), Color.White)
        b.ForeColor = If(active, Color.White, If(enabled, Color.Black, Color.Silver))
        b.Enabled = enabled

        If enabled Then
            AddHandler b.Click, Sub()
                                    profPage = page
                                    ShowProfPage()
                                End Sub
        End If
        Return b
    End Function

    ' ---------- drawing helpers ----------
    Private Function RoundedRect(r As Rectangle, radius As Integer) As GraphicsPath
        Dim gp As New GraphicsPath()
        Dim d As Integer = radius * 2
        gp.AddArc(r.X, r.Y, d, d, 180, 90)
        gp.AddArc(r.Right - d, r.Y, d, d, 270, 90)
        gp.AddArc(r.Right - d, r.Bottom - d, d, d, 0, 90)
        gp.AddArc(r.X, r.Bottom - d, d, d, 90, 90)
        gp.CloseFigure()
        Return gp
    End Function

    Private Sub DrawPill(g As Graphics, r As Rectangle, back As Color, fore As Color, text As String)
        Using path = RoundedRect(r, 8), br As New SolidBrush(back), tb As New SolidBrush(fore)
            g.FillPath(br, path)
            Dim sf As New StringFormat() With {.Alignment = StringAlignment.Center, .LineAlignment = StringAlignment.Center}
            g.DrawString(text, pillFont, tb, r, sf)
        End Using
    End Sub

    ' Edit / Delete rectangles inside a cell
    Private Function ActionRects(cell As Rectangle) As Rectangle()
        Const w As Integer = 52, h As Integer = 24, gap As Integer = 6
        Dim startX As Integer = cell.X + (cell.Width - (w * 2 + gap)) \ 2
        Dim y As Integer = cell.Y + (cell.Height - h) \ 2
        Return {New Rectangle(startX, y, w, h), New Rectangle(startX + w + gap, y, w, h)}
    End Function

    Private Sub dgvProf_CellPainting(sender As Object, e As DataGridViewCellPaintingEventArgs) Handles dgvProf.CellPainting
        If e.RowIndex < 0 Then Return
        Dim col As String = dgvProf.Columns(e.ColumnIndex).Name
        If col <> "colStatus" AndAlso col <> "colActions" Then Return

        e.PaintBackground(e.ClipBounds, False)
        e.Graphics.SmoothingMode = SmoothingMode.AntiAlias

        If col = "colStatus" Then
            Dim status As String = Convert.ToString(e.Value)
            Dim r As New Rectangle(e.CellBounds.X + (e.CellBounds.Width - 80) \ 2,
                                   e.CellBounds.Y + (e.CellBounds.Height - 24) \ 2, 80, 24)
            If status = "Active" Then
                DrawPill(e.Graphics, r, Color.FromArgb(198, 240, 214), Color.FromArgb(0, 100, 0), "Active")
            Else
                DrawPill(e.Graphics, r, Color.FromArgb(250, 190, 190), Color.FromArgb(180, 0, 0), "Inactive")
            End If
        Else
            Dim rects = ActionRects(e.CellBounds)
            DrawPill(e.Graphics, rects(0), Color.FromArgb(198, 240, 214), Color.FromArgb(0, 100, 0), "Edit")
            DrawPill(e.Graphics, rects(1), Color.FromArgb(250, 190, 190), Color.FromArgb(180, 0, 0), "Delete")
        End If

        e.Handled = True
    End Sub

    Private Sub dgvProf_CellMouseClick(sender As Object, e As DataGridViewCellMouseEventArgs) Handles dgvProf.CellMouseClick
        If e.RowIndex < 0 Then Return
        If dgvProf.Columns(e.ColumnIndex).Name <> "colActions" Then Return

        Dim size = dgvProf.GetCellDisplayRectangle(e.ColumnIndex, e.RowIndex, False).Size
        Dim rects = ActionRects(New Rectangle(0, 0, size.Width, size.Height))
        Dim empId As String = dgvProf.Rows(e.RowIndex).Cells("colEmpId").Value.ToString()

        If rects(0).Contains(e.Location) Then
            MessageBox.Show("Edit " & empId)             ' open your edit form here
        ElseIf rects(1).Contains(e.Location) Then
            If MessageBox.Show("Delete " & empId & "?", "Confirm", MessageBoxButtons.YesNo) = DialogResult.Yes Then
                allProfs.RemoveAll(Function(p) p.EmployeeID = empId)
                ApplyProfFilter()
            End If
        End If
    End Sub

Imports System.Drawing
Imports System.Windows.Forms
Imports System.ComponentModel
Imports System.Drawing.Drawing2D

Public Class CustomButton
    Inherits Button

    Private _borderColor As Color = Color.Black
    Private _borderSize As Integer = 2
    Private _cornerRadius As Integer = 10
    Private _hoverColor As Color = Color.LightGray
    Private _normalColor As Color = Color.LightBlue

    <Category("Appearance"), Description("Border color of the button.")>
    Public Property BorderColor As Color
        Get
            Return _borderColor
        End Get
        Set(value As Color)
            _borderColor = value
            Me.Invalidate()
        End Set
    End Property

    <Category("Appearance"), Description("Border thickness of the button.")>
    Public Property BorderSize As Integer
        Get
            Return _borderSize
        End Get
        Set(value As Integer)
            _borderSize = value
            Me.Invalidate()
        End Set
    End Property

    <Category("Appearance"), Description("Corner radius for rounded edges.")>
    Public Property CornerRadius As Integer
        Get
            Return _cornerRadius
        End Get
        Set(value As Integer)
            _cornerRadius = value
            Me.Invalidate()
        End Set
    End Property

    <Category("Appearance"), Description("Background color when hovered.")>
    Public Property HoverColor As Color
        Get
            Return _hoverColor
        End Get
        Set(value As Color)
            _hoverColor = value
        End Set
    End Property

    <Category("Appearance"), Description("Normal background color.")>
    Public Property NormalColor As Color
        Get
            Return _normalColor
        End Get
        Set(value As Color)
            _normalColor = value
            Me.BackColor = value
            Me.Invalidate()
        End Set
    End Property

    Protected Overrides Sub OnPaint(pevent As PaintEventArgs)
        Dim g As Graphics = pevent.Graphics
        g.SmoothingMode = SmoothingMode.AntiAlias

        Dim rect As Rectangle = Me.ClientRectangle
        Using path As GraphicsPath = GetRoundedRectPath(rect, _cornerRadius)
            Me.Region = New Region(path)

            ' Fill background
            Using brush As New SolidBrush(Me.BackColor)
                g.FillPath(brush, path)
            End Using

            ' Draw border
            Using pen As New Pen(_borderColor, _borderSize)
                g.DrawPath(pen, path)
            End Using
        End Using

        ' Draw text centered
        Dim sf As New StringFormat() With {
            .Alignment = StringAlignment.Center,
            .LineAlignment = StringAlignment.Center
        }
        Using textBrush As New SolidBrush(Me.ForeColor)
            g.DrawString(Me.Text, Me.Font, textBrush, rect, sf)
        End Using
    End Sub

    Protected Overrides Sub OnMouseEnter(e As EventArgs)
        MyBase.OnMouseEnter(e)
        Me.BackColor = _hoverColor
    End Sub

    Protected Overrides Sub OnMouseLeave(e As EventArgs)
        MyBase.OnMouseLeave(e)
        Me.BackColor = _normalColor
    End Sub

    Private Function GetRoundedRectPath(rect As Rectangle, radius As Integer) As GraphicsPath
        Dim path As New GraphicsPath()
        path.AddArc(rect.X, rect.Y, radius, radius, 180, 90)
        path.AddArc(rect.Right - radius, rect.Y, radius, radius, 270, 90)
        path.AddArc(rect.Right - radius, rect.Bottom - radius, radius, radius, 0, 90)
        path.AddArc(rect.X, rect.Bottom - radius, radius, radius, 90, 90)
        path.CloseFigure()
        Return path
    End Function
End Class

Imports System.ComponentModel
Imports System.Drawing.Drawing2D

Public Class CustomPanel
    Inherits Panel

    Private _cornerRadius As Integer = 20
    Private _borderColor As Color = Color.FromArgb(0, 85, 0)
    Private _borderThickness As Integer = 2

    <Category("Appearance"), DefaultValue(20)>
    Public Property CornerRadius As Integer
        Get
            Return _cornerRadius
        End Get
        Set(value As Integer)
            _cornerRadius = Math.Max(0, value)
            Invalidate()
        End Set
    End Property

    <Category("Appearance")>
    Public Property BorderColor As Color
        Get
            Return _borderColor
        End Get
        Set(value As Color)
            _borderColor = value
            Invalidate()
        End Set
    End Property

    <Category("Appearance"), DefaultValue(2)>
    Public Property BorderThickness As Integer
        Get
            Return _borderThickness
        End Get
        Set(value As Integer)
            _borderThickness = Math.Max(0, value)
            Invalidate()
        End Set
    End Property

    Private _clipCorners As Boolean = False

    <Category("Appearance"), DefaultValue(False)>
    Public Property ClipCorners As Boolean
        Get
            Return _clipCorners
        End Get
        Set(value As Boolean)
            _clipCorners = value
            UpdateRegion()
            Invalidate()
        End Set
    End Property

    Protected Overrides Sub OnResize(e As EventArgs)
        MyBase.OnResize(e)
        UpdateRegion()
    End Sub

    Private Sub UpdateRegion()
        If _clipCorners AndAlso Width > 0 AndAlso Height > 0 Then
            Using path As GraphicsPath = RoundedRect(New Rectangle(0, 0, Width, Height), _cornerRadius)
                Region = New Region(path)
            End Using
        Else
            Region = Nothing
        End If
    End Sub

    Public Sub New()
        SetStyle(ControlStyles.AllPaintingInWmPaint Or
                 ControlStyles.OptimizedDoubleBuffer Or
                 ControlStyles.ResizeRedraw Or
                 ControlStyles.UserPaint, True)
        BackColor = Color.White
    End Sub

    ' Paint the corners with the parent's color so they look transparent
    Protected Overrides Sub OnPaintBackground(e As PaintEventArgs)
        Dim c As Color = If(Parent IsNot Nothing, Parent.BackColor, SystemColors.Control)
        e.Graphics.Clear(c)
    End Sub

    Protected Overrides Sub OnPaint(e As PaintEventArgs)
        MyBase.OnPaint(e)
        e.Graphics.SmoothingMode = SmoothingMode.AntiAlias

        Dim half As Integer = _borderThickness \ 2
        Dim rect As New Rectangle(half, half,
                                  Width - _borderThickness - 1,
                                  Height - _borderThickness - 1)

        Using path As GraphicsPath = RoundedRect(rect, _cornerRadius)
            Using fill As New SolidBrush(BackColor)
                e.Graphics.FillPath(fill, path)
            End Using
            If _borderThickness > 0 Then
                Using pen As New Pen(_borderColor, _borderThickness)
                    e.Graphics.DrawPath(pen, path)
                End Using
            End If
        End Using
    End Sub

    Public Shared Function RoundedRect(r As Rectangle, radius As Integer) As GraphicsPath
        Dim path As New GraphicsPath()
        If radius <= 0 Then
            path.AddRectangle(r)
            Return path
        End If
        Dim d As Integer = radius * 2
        path.AddArc(r.X, r.Y, d, d, 180, 90)
        path.AddArc(r.Right - d, r.Y, d, d, 270, 90)
        path.AddArc(r.Right - d, r.Bottom - d, d, d, 0, 90)
        path.AddArc(r.X, r.Bottom - d, d, d, 90, 90)
        path.CloseFigure()
        Return path
    End Function

End Class

Public Class CustomProgressBar
    Inherits Control

    Private _barColor As Color = Color.LightGreen
    Private _borderColor As Color = Color.Black
    Private _cornerRadius As Integer = 10

    Private _textColor As Color = Color.Black
    Private _textAlign As ContentAlignment = ContentAlignment.MiddleCenter

    <Category("Behavior"), Description("Current progress value.")>
    Public Property Value As Integer

            Return _value
        End Get

    End Property

    <Category("Behavior"), Description("Maximum progress value.")>
    Public Property Maximum As Integer
        Get
            Return _maximum
        End Get

            _maximum = v

    End Property

    Public Property BarColor As Color

            Return _barColor

            Return _cornerRadius
        End Get
        Set(v As Integer)
            _cornerRadius = v
            Me.Invalidate()
        End Set
    End Property

    <Category("Appearance"), Description("Show percentage text inside the bar.")>
    Public Property ShowText As Boolean
        Get
            Return _showText
        End Get
        Set(v As Boolean)
            _showText = v
            Me.Invalidate()
        End Set
    End Property

    <Category("Appearance"), Description("Color of the percentage text.")>
    Public Property TextColor As Color
        Get
            Return _textColor
        End Get

        End Set
    End Property

            Return _textAlign
        End Get

            Me.Invalidate()
        End Set

        MyBase.OnPaint(e)

        Dim g As Graphics = e.Graphics
        g.SmoothingMode = SmoothingMode.AntiAlias
        Dim rect As Rectangle = Me.ClientRectangle

        ' Border
        Using borderPen As New Pen(_borderColor, 2)
            Using borderPath As GraphicsPath = GetRoundedRectPath(rect, _cornerRadius)
                g.DrawPath(borderPen, borderPath)
            End Using
        End Using

        ' Progress fill
        Dim progressWidth As Integer = CInt((rect.Width - 4) * (_value / _maximum))
        If progressWidth > 0 Then
            Dim progressRect As New Rectangle(rect.X + 2, rect.Y + 2, progressWidth, rect.Height - 4)
            Using brush As New SolidBrush(_barColor)
                g.FillRectangle(brush, progressRect)
            End Using
        End If

        ' Text
        If _showText Then
            Dim percent As Integer = CInt((_value / _maximum) * 100)
            Dim text As String = percent.ToString() & "%"
            Dim format As New StringFormat()

            Select Case _textAlign
                Case ContentAlignment.TopLeft
                    format.Alignment = StringAlignment.Near
                    format.LineAlignment = StringAlignment.Near
                Case ContentAlignment.TopCenter
                    format.Alignment = StringAlignment.Center
                    format.LineAlignment = StringAlignment.Near
                Case ContentAlignment.TopRight
                    format.Alignment = StringAlignment.Far
                    format.LineAlignment = StringAlignment.Near
                Case ContentAlignment.MiddleLeft
                    format.Alignment = StringAlignment.Near
                    format.LineAlignment = StringAlignment.Center
                Case ContentAlignment.MiddleCenter
                    format.Alignment = StringAlignment.Center
                    format.LineAlignment = StringAlignment.Center
                Case ContentAlignment.MiddleRight
                    format.Alignment = StringAlignment.Far
                    format.LineAlignment = StringAlignment.Center
                Case ContentAlignment.BottomLeft
                    format.Alignment = StringAlignment.Near
                    format.LineAlignment = StringAlignment.Far
                Case ContentAlignment.BottomCenter
                    format.Alignment = StringAlignment.Center
                    format.LineAlignment = StringAlignment.Far
                Case ContentAlignment.BottomRight
                    format.Alignment = StringAlignment.Far
                    format.LineAlignment = StringAlignment.Far
            End Select

            Using textBrush As New SolidBrush(_textColor)
                g.DrawString(text, Me.Font, textBrush, rect, format)
            End Using
        End If
    End Sub

    Private Function GetRoundedRectPath(rect As Rectangle, radius As Integer) As GraphicsPath
        Dim path As New GraphicsPath()
        path.AddArc(rect.X, rect.Y, radius, radius, 180, 90)
        path.AddArc(rect.Right - radius, rect.Y, radius, radius, 270, 90)
        path.AddArc(rect.Right - radius, rect.Bottom - radius, radius, radius, 0, 90)
        path.AddArc(rect.X, rect.Bottom - radius, radius, radius, 90, 90)
        path.CloseFigure()
        Return path
    End Function

Public Class frmlogin

    Private ReadOnly authService As New AuthService()

    Private Sub frmLogin_Load(sender As Object, e As EventArgs) Handles MyBase.Load
        ' Mask password characters in TextBox1
        TextBox1.UseSystemPasswordChar = True

        ' Apply background opacity effect
        If Me.BackgroundImage IsNot Nothing Then
            Dim original As Image = Me.BackgroundImage
            Me.BackgroundImage = FadeImage(original, 0.5F)
        End If
    End Sub

    ' Opacity helper
    Private Function FadeImage(src As Image, opacity As Single) As Image
        Dim bmp As New Bitmap(src.Width, src.Height)
        Using g As Graphics = Graphics.FromImage(bmp)
            g.Clear(Color.White)
            Dim cm As New Imaging.ColorMatrix() With {.Matrix33 = opacity}
            Using ia As New Imaging.ImageAttributes()
                ia.SetColorMatrix(cm)
                g.DrawImage(src, New Rectangle(0, 0, bmp.Width, bmp.Height),
                        0, 0, src.Width, src.Height, GraphicsUnit.Pixel, ia)
            End Using
        End Using
        Return bmp
    End Function

    ' LOG IN BUTTON CLICK (Button1)
    Private Sub Button1_Click(sender As Object, e As EventArgs) Handles Button1.Click
        Dim username As String = ValidationHelper.SanitizeInput(boxuser.Text)
        Dim password As String = TextBox1.Text.Trim()


        If Not ValidationHelper.IsRequired(username) OrElse Not ValidationHelper.IsRequired(password) Then
            MessageBox.Show("Please enter both your Username/ID and Password.", "Validation Error", MessageBoxButtons.OK, MessageBoxIcon.Warning)
            boxuser.Focus()
            Return
        End If

        Try
            ' 2. Authenticate user against database via UserService
            Dim user As UserModel = authService.VerifyLogin(username, password)

            If user IsNot Nothing Then

                Dim role As String = user.Role.Trim().ToLower()
                Dim picked As String = If(RadioButton1.Checked, "student", "teacher")

                ' Admins have no radio button, so let them through with either selected
                If role <> "admin" AndAlso role <> picked Then
                    MessageBox.Show("This account is not a " & picked & " account. Please select the correct role.",
                    "Wrong role", MessageBoxButtons.OK, MessageBoxIcon.Warning)
                    TextBox1.Clear()
                    Return
                End If

                ' 3. Save user state in AppSession
                AppSession.StartSession(user)

                ' 4. Route to designated form based on database user role
                Me.Hide()

                Select Case user.Role.Trim().ToLower()
                    Case "admin"
                        Dim adminForm As New Admin()
                        adminForm.Show()

                    Case "teacher", "professor", "prof"
                        Dim profForm As New frmprof()
                        profForm.Show()

                    Case "student"
                        Dim studentForm As New frmstudent()
                        studentForm.Show()

                    Case Else
                        MessageBox.Show("Unrecognized user role assigned.", "Access Denied", MessageBoxButtons.OK, MessageBoxIcon.Error)
                        AppSession.ClearSession()
                        Me.Show()
                End Select
            Else
                MessageBox.Show("Invalid Username/ID or Password.", "Login Failed", MessageBoxButtons.OK, MessageBoxIcon.Error)
                TextBox1.Clear()
                TextBox1.Focus()
            End If

        Catch ex As Exception
            MessageBox.Show($"Database connection error: {ex.Message}", "System Error", MessageBoxButtons.OK, MessageBoxIcon.Error)
        End Try
    End Sub

    Private Sub SetAllComboBoxes(parent As Control)

            If TypeOf ctrl Is ComboBox Then

                If cb.Items.Count > 0 Then
                    cb.SelectedIndex = 0
                End If
            End If

            ' If the control contains children (Panel, GroupBox, TabPage, etc.), check inside too
            If ctrl.HasChildren Then
                SetAllComboBoxes(ctrl)
            End If

    Private Sub AlignGridToHeader()

        Dim cSW1 As Integer = HeaderCenterX(cboSW1)
        Dim cHW1 As Integer = HeaderCenterX(cboHW1)
        Dim cQuiz As Integer = HeaderCenterX(cboQuiz1)
        Dim cMid As Integer = HeaderCenterX(lblMidterms)
        Dim cRun As Integer = HeaderCenterX(lblRunningGrade)
        Dim cStat As Integer = HeaderCenterX(lblStatus)
        Dim gap As Double = (cMid - cQuiz) / 4

        Dim names() As String = {"Student", "SW1", "HW1", "Quiz1", "Act4", "Act5", "Act6",
                                 "Midterms", "Running Grade", "Status"}
        Dim centers() As Double = {0, cSW1, cHW1, cQuiz,
                                   cQuiz + gap, cQuiz + 2 * gap, cQuiz + 3 * gap,
                                   cMid, cRun, cStat}

        Dim startX As Double = 0
        For i As Integer = 0 To names.Length - 2          ' every column except Status
            Dim endX As Double
            If i = 0 Then
                endX = cSW1 - 55                          ' Student column width
            Else
                endX = (centers(i) + centers(i + 1)) / 2
            End If

            With dgvClassGrades.Columns(names(i))
                .AutoSizeMode = DataGridViewAutoSizeColumnMode.None
                .Width = CInt(Math.Max(endX - startX, 40))
                .DefaultCellStyle.Alignment = If(i = 0,
                    DataGridViewContentAlignment.MiddleLeft,
                    DataGridViewContentAlignment.MiddleCenter)
            End With
            startX = endX
        Next

        ' Status takes whatever space is left, so no horizontal scrollbar
        With dgvClassGrades.Columns("Status")
            .AutoSizeMode = DataGridViewAutoSizeColumnMode.Fill
            .DefaultCellStyle.Alignment = DataGridViewContentAlignment.MiddleCenter
        End With

    ' Sizes the two cards to fill the page and spreads the four tiles in one row
    Private Sub LayoutAnalytics()
        Const margin As Integer = 20
        Const gap As Integer = 20
        Const minCardH As Integer = 400        ' never shorter than this (normal window size)
        Const maxHeightRatio As Double = 0.5   ' 0.5 = cards end at the middle of the page

        ' visible height = bottom of the screen's working area minus where the form's client area starts
        Dim clientTopOnScreen As Integer = Me.PointToScreen(Point.Empty).Y
        Dim visibleH As Integer = Math.Min(Me.ClientSize.Height,
            Screen.FromControl(Me).WorkingArea.Bottom - clientTopOnScreen)

        pnlAnalytics.SetBounds(pnlAnalytics.Left, pnlAnalytics.Top,
                               Me.ClientSize.Width - pnlAnalytics.Left,
                               visibleH - pnlAnalytics.Top)

        Dim cardW As Integer = (pnlAnalytics.ClientSize.Width - 2 * margin - gap) \ 2

        ' height: stop at the middle of the page, but never below minCardH
        ' and never taller than the space that is actually available
        Dim availH As Integer = pnlAnalytics.ClientSize.Height - RoundedPanel4.Top - margin
        Dim cardH As Integer = CInt(pnlAnalytics.ClientSize.Height * maxHeightRatio) - RoundedPanel4.Top
        cardH = Math.Min(Math.Max(cardH, minCardH), availH)

        If cardW < 300 OrElse cardH < 200 Then Exit Sub

        RoundedPanel4.SetBounds(margin, RoundedPanel4.Top, cardW, cardH)
        RoundedPanel5.SetBounds(margin + cardW + gap, RoundedPanel4.Top, cardW, cardH)

        ' four tiles in one row inside the left card
        Const tMargin As Integer = 12
        Const tGap As Integer = 12
        Dim tiles() As Control = {RoundedPanel6, RoundedPanel7, RoundedPanel8, RoundedPanel9}
        Dim tileW As Integer = (RoundedPanel4.ClientSize.Width - 2 * tMargin - 3 * tGap) \ 4
        If tileW < 80 Then Exit Sub

        Dim tileTop As Integer = RoundedPanel6.Top
        Dim tileH As Integer = RoundedPanel6.Height
        For i As Integer = 0 To tiles.Length - 1
            tiles(i).SetBounds(tMargin + i * (tileW + tGap), tileTop, tileW, tileH)
        Next
    End Sub

    ' Creates one big number label inside a tile
    Private Function AddValueLabel(tile As Control, textColor As Color) As Label
        Dim l As New Label()
        l.AutoSize = False
        l.BackColor = Color.Transparent
        l.ForeColor = textColor
        l.Font = New Font("Segoe UI", 20.0F, FontStyle.Bold)
        l.TextAlign = ContentAlignment.MiddleCenter
        l.SetBounds(0, 10, tile.Width, 55)
        l.Anchor = AnchorStyles.Top Or AnchorStyles.Left Or AnchorStyles.Right
        tile.Controls.Add(l)
        l.BringToFront()
        Return l
    End Function

        Dim green As Color = Color.FromArgb(0, 100, 0)

                    cap.AutoSize = False
                    cap.Dock = DockStyle.Bottom

    ' Weighted average of graded items only
    Private Sub UpdateStanding()
        Dim total As Double = 0
        Dim weights As Double = 0

        For Each row As DataGridViewRow In dgvgrades.Rows
            Dim parts = Convert.ToString(row.Cells(1).Value).Split("/"c)
            If parts.Length <> 2 Then Continue For   ' skips ungraded rows like "---"

            Dim score, maxScore, weight As Double
            If Double.TryParse(parts(0), score) AndAlso
               Double.TryParse(parts(1), maxScore) AndAlso maxScore > 0 AndAlso
               Double.TryParse(Convert.ToString(row.Cells(2).Value).TrimEnd("%"c), weight) Then
                total += score / maxScore * 100 * weight
                weights += weight
            End If
        Next

        If weights > 0 Then
            Dim pct As Double = Math.Round(total / weights, 1)
            ' 75 is an assumed passing mark. Use your school's.
            lblstanding.Text = "Current Standing: " & pct & "% " & If(pct >= 75, "(Passing)", "(Failing)")
        End If
    End Sub

    ' Each stat card has two labels. This finds the pair inside the card of
    ' "anyLabel", puts the number on the one with the bigger font and the
    ' caption on the other, so it works whichever label is named what.
    Private Sub SetStat(anyLabel As Label, value As String, caption As String)
        Dim other As Label = Nothing
        If anyLabel.Parent IsNot Nothing Then
            For Each c As Control In anyLabel.Parent.Controls
                If TypeOf c Is Label AndAlso c IsNot anyLabel Then
                    other = DirectCast(c, Label)
                    Exit For
                End If
            Next
        End If

        If other Is Nothing Then
            anyLabel.Text = value   ' card only has one label
            Return
        End If

        Dim bigLabel As Label = If(anyLabel.Font.Size >= other.Font.Size, anyLabel, other)
        Dim smallLabel As Label = If(bigLabel Is anyLabel, other, anyLabel)

        bigLabel.Text = value
        smallLabel.Text = caption
    End Sub

Imports System.Drawing
Imports System.Drawing.Drawing2D
Imports System.Windows.Forms

Public Class PillLabel
    Inherits Label

    Private _cornerRadius As Integer = 20
    Public Property CornerRadius As Integer
        Get
            Return _cornerRadius
        End Get
        Set(value As Integer)
            _cornerRadius = value
            Me.Invalidate() ' Redraw when radius changes
        End Set
    End Property

    Protected Overrides Sub OnPaint(e As PaintEventArgs)
        MyBase.OnPaint(e)

        ' Create rounded rectangle path
        Dim path As New GraphicsPath()
        Dim rect As Rectangle = Me.ClientRectangle
        Dim radius As Integer = _cornerRadius

        path.AddArc(rect.X, rect.Y, radius, radius, 180, 90)
        path.AddArc(rect.Right - radius, rect.Y, radius, radius, 270, 90)
        path.AddArc(rect.Right - radius, rect.Bottom - radius, radius, radius, 0, 90)
        path.AddArc(rect.X, rect.Bottom - radius, radius, radius, 90, 90)
        path.CloseFigure()

        Me.Region = New Region(path)
    End Sub
End Class

Imports System.Drawing.Drawing2D
Imports System.Windows.Forms

Public Class RoundedPanel
    Inherits Panel

    ' How round the corners are
    Public Property CornerRadius As Integer = 20

    ' Border customization
    Public Property BorderColor As Color = Color.DarkGreen
    Public Property BorderThickness As Integer = 2

    Protected Overrides Sub OnPaint(e As PaintEventArgs)
        MyBase.OnPaint(e)

        Dim rect As Rectangle = Me.ClientRectangle
        rect.Width -= 1
        rect.Height -= 1

        Using path As GraphicsPath = GetRoundedPath(rect, CornerRadius)
            e.Graphics.SmoothingMode = SmoothingMode.AntiAlias
            e.Graphics.FillPath(New SolidBrush(Me.BackColor), path)
            e.Graphics.DrawPath(New Pen(BorderColor, BorderThickness), path)
        End Using
    End Sub

    Private Function GetRoundedPath(rect As Rectangle, radius As Integer) As GraphicsPath
        Dim path As New GraphicsPath()
        path.AddArc(rect.X, rect.Y, radius, radius, 180, 90)
        path.AddArc(rect.Right - radius, rect.Y, radius, radius, 270, 90)
        path.AddArc(rect.Right - radius, rect.Bottom - radius, radius, radius, 0, 90)
        path.AddArc(rect.X, rect.Bottom - radius, radius, radius, 90, 90)
        path.CloseFigure()
        Return path
    End Function
End Class

' ===== DatabaseHelper.vb =====

Imports System.Data
Imports System.IO
Imports System.Text.Json
Imports MySql.Data.MySqlClient

Public Class DatabaseHelper

    Private Shared _connString As String
    Private Shared ReadOnly LockObj As New Object()

    Private Shared Function GetConnString() As String
        If _connString Is Nothing Then
            SyncLock LockObj
                If _connString Is Nothing Then
                    _connString = BuildConnString()
                End If
            End SyncLock
        End If
        Return _connString
    End Function

    Private Shared Function BuildConnString() As String
        Dim host As String = "127.0.0.1"
        Dim port As Integer = 3306
        Dim db As String = "transparent_grading_system"
        Dim user As String = "root"
        Dim pass As String = ""

        Dim configPath As String = Path.Combine(AppContext.BaseDirectory, "dbconfig.json")
        If File.Exists(configPath) Then
            Try
                Using doc As JsonDocument = JsonDocument.Parse(File.ReadAllText(configPath))
                    Dim root As JsonElement = doc.RootElement
                    host = ReadText(root, "Host", host)
                    db = ReadText(root, "Database", db)
                    user = ReadText(root, "User", user)
                    pass = ReadText(root, "Password", pass)

                    Dim portEl As JsonElement
                    If root.TryGetProperty("Port", portEl) AndAlso portEl.ValueKind = JsonValueKind.Number Then
                        port = portEl.GetInt32()
                    End If
                End Using
            Catch ex As JsonException
                Throw New InvalidOperationException("dbconfig.json is not valid JSON: " & ex.Message)
            End Try
        End If

        Dim b As New MySqlConnectionStringBuilder()
        b.Server = host
        b.Port = CUInt(port)
        b.Database = db
        b.UserID = user
        b.Password = pass
        b.SslMode = MySqlSslMode.Preferred
        b.AllowZeroDateTime = True
        Return b.ConnectionString
    End Function

    Private Shared Function ReadText(root As JsonElement, key As String, fallback As String) As String
        Dim el As JsonElement
        If root.TryGetProperty(key, el) AndAlso el.ValueKind = JsonValueKind.String Then
            Return el.GetString()
        End If
        Return fallback
    End Function

    Public Shared Function GetConnection() As MySqlConnection
        Dim conn As New MySqlConnection(GetConnString())
        conn.Open()
        Return conn
    End Function


    Public Shared Function ExecuteQuery(sql As String, ParamArray params As MySqlParameter()) As DataTable
        Dim dt As New DataTable()
        Using conn As MySqlConnection = GetConnection()
            Using cmd As New MySqlCommand(sql, conn)
                If params IsNot Nothing AndAlso params.Length > 0 Then
                    cmd.Parameters.AddRange(params)
                End If
                Using adapter As New MySqlDataAdapter(cmd)
                    adapter.Fill(dt)
                End Using
            End Using
        End Using
        Return dt
    End Function

    Public Shared Function ExecuteNonQuery(sql As String, ParamArray params As MySqlParameter()) As Integer
        Using conn As MySqlConnection = GetConnection()
            Using cmd As New MySqlCommand(sql, conn)
                If params IsNot Nothing AndAlso params.Length > 0 Then
                    cmd.Parameters.AddRange(params)
                End If
                Return cmd.ExecuteNonQuery()
            End Using
        End Using
    End Function

    Public Shared Function ExecuteScalar(sql As String, ParamArray params As MySqlParameter()) As Object
        Using conn As MySqlConnection = GetConnection()
            Using cmd As New MySqlCommand(sql, conn)
                If params IsNot Nothing AndAlso params.Length > 0 Then
                    cmd.Parameters.AddRange(params)
                End If
                Return cmd.ExecuteScalar()
            End Using
        End Using

' ===== PasswordHasher.vb =====

Imports System.Security.Cryptography
Imports System.Text

Public Class PasswordHasher

    ''' <summary>
    ''' Verifies a raw password against a stored database hash or plain-text record.
    ''' </summary>
    Public Shared Function VerifyPassword(rawPassword As String, storedHash As String) As Boolean
        If String.IsNullOrWhiteSpace(storedHash) OrElse String.IsNullOrWhiteSpace(rawPassword) Then
            Return False
        End If

        ' 1. Plain-text match (for testing / unhashed seed data)
        If rawPassword = storedHash Then Return True

        ' 2. Built-in SHA256 hash verification
        Dim hashedInput As String = HashSHA256(rawPassword)
        If hashedInput.Equals(storedHash, StringComparison.OrdinalIgnoreCase) Then Return True

        Return False
    End Function

    Public Shared Function HashSHA256(input As String) As String
        Using sha256 As SHA256 = sha256.Create()
            Dim bytes As Byte() = sha256.ComputeHash(Encoding.UTF8.GetBytes(input))
            Dim sb As New StringBuilder()
            For Each b As Byte In bytes
                sb.Append(b.ToString("x2"))
            Next
            Return sb.ToString()
        End Using
    End Function

End Class

Imports MySql.Data.MySqlClient
Imports System.Data

Public Class AuditService

    ' Standalone: opens its own connection
    Public Shared Function LogAction(actorRole As String, actorUserId As String, action As String,
                                     entityType As String, entityId As String,
                                     Optional oldValue As String = Nothing,
                                     Optional newValue As String = Nothing,
                                     Optional details As String = Nothing) As Boolean
        Try
            Using conn As MySqlConnection = DatabaseHelper.GetConnection()
                Return InsertLog(conn, Nothing, actorRole, actorUserId, action, entityType, entityId, oldValue, newValue, details)
            End Using
        Catch ex As Exception
            Console.WriteLine("[AUDIT LOG ERROR] " & ex.Message)
            Return False
        End Try
    End Function

    ' Inside an existing transaction (so the log rolls back together with the grade change)
    Public Shared Function LogActionTx(conn As MySqlConnection, tx As MySqlTransaction,
                                       actorRole As String, actorUserId As String, action As String,
                                       entityType As String, entityId As String,
                                       Optional oldValue As String = Nothing,
                                       Optional newValue As String = Nothing,
                                       Optional details As String = Nothing) As Boolean
        Return InsertLog(conn, tx, actorRole, actorUserId, action, entityType, entityId, oldValue, newValue, details)
    End Function

    Private Shared Function InsertLog(conn As MySqlConnection, tx As MySqlTransaction,
                                      actorRole As String, actorUserId As String, action As String,
                                      entityType As String, entityId As String,
                                      oldValue As String, newValue As String, details As String) As Boolean
        Dim studentId As Object = DBNull.Value
        Dim teacherId As Object = DBNull.Value
        Dim adminId As Object = DBNull.Value

        Select Case actorRole.Trim().ToLower()
            Case "student"
                studentId = actorUserId
            Case "teacher"
                teacherId = actorUserId
            Case "admin"
                Dim parsed As Integer
                If Not Integer.TryParse(actorUserId, parsed) Then
                    Throw New ArgumentException("Admin id must be numeric.")
                End If
                adminId = parsed
            Case Else
                Throw New ArgumentException("Unknown actor role: " & actorRole)
        End Select

        Dim sql As String =
            "INSERT INTO auditlog (student_id, teacher_id, admin_id, action, entity_type, entity_id, old_value, new_value, details) " &
            "VALUES (@StudentId, @TeacherId, @AdminId, @Action, @EntityType, @EntityId, @OldVal, @NewVal, @Details)"

        Using cmd As New MySqlCommand(sql, conn, tx)
            cmd.Parameters.AddWithValue("@StudentId", studentId)
            cmd.Parameters.AddWithValue("@TeacherId", teacherId)
            cmd.Parameters.AddWithValue("@AdminId", adminId)
            cmd.Parameters.AddWithValue("@Action", Clip(action, 100))
            cmd.Parameters.AddWithValue("@EntityType", Clip(entityType, 50))
            cmd.Parameters.AddWithValue("@EntityId", Clip(entityId, 20))
            cmd.Parameters.AddWithValue("@OldVal", OrNull(oldValue, 100))
            cmd.Parameters.AddWithValue("@NewVal", OrNull(newValue, 100))
            cmd.Parameters.AddWithValue("@Details", OrNull(details, 500))
            Return cmd.ExecuteNonQuery() > 0
        End Using
    End Function

    ' Trim to the column length so the insert doesn't fail with "Data too long"
    Private Shared Function Clip(s As String, maxLen As Integer) As String
        If s Is Nothing Then Return ""
        Return If(s.Length > maxLen, s.Substring(0, maxLen), s)
    End Function

    Private Shared Function OrNull(s As String, maxLen As Integer) As Object
        If String.IsNullOrEmpty(s) Then Return DBNull.Value
        Return Clip(s, maxLen)
    End Function

    Public Shared Function GetLogsByActor(actorRole As String, actorUserId As String) As DataTable
        Dim col As String
        Select Case actorRole.Trim().ToLower()
            Case "student" : col = "student_id"
            Case "teacher" : col = "teacher_id"
            Case "admin" : col = "admin_id"
            Case Else : Return New DataTable()
        End Select

        Dim sql As String =
            "SELECT log_id, student_id, teacher_id, admin_id, action, entity_type, entity_id, old_value, new_value, timestamp, details " &
            "FROM auditlog WHERE " & col & " = @ActorId ORDER BY timestamp DESC"

        Return DatabaseHelper.ExecuteQuery(sql, {New MySqlParameter("@ActorId", actorUserId)})
    End Function

        ' Cast IDs to CHAR to prevent MySQL UNION data type mismatch conflicts
        Dim sql As String =
            "SELECT 'Admin' AS Role, CAST(admin_id AS CHAR) AS UserId, full_name AS FullName, email, password_hash " &
            "FROM admin WHERE email = @Login " &
            "UNION ALL " &
            "SELECT 'Teacher', CAST(teacher_id AS CHAR), full_name, email, password_hash " &
            "FROM teacher WHERE (email = @Login OR teacher_id = @Login) AND status = 'Active' " &
            "UNION ALL " &
            "SELECT 'Student', CAST(student_id AS CHAR), full_name, email, password_hash " &
            "FROM student WHERE (email = @Login OR student_id = @Login) AND status = 'Active'"

        Dim params As MySqlParameter() = {New MySqlParameter("@Login", cleanLogin)}
        Dim dt As DataTable = DatabaseHelper.ExecuteQuery(sql, params)

        For Each row As DataRow In dt.Rows
            If SecurityHelper.VerifyPassword(rawPassword, row("password_hash").ToString()) Then

                    .UserId = row("UserId").ToString(),
                    .FullName = row("FullName").ToString(),
                    .Email = row("email").ToString(),
                    .Role = row("Role").ToString()
                }
            End If

                        rowFields.Add($"""{cellValue}""")
                    Next
                    writer.WriteLine(String.Join(",", rowFields))
                Next
            End Using

        Catch ex As Exception
            Console.WriteLine("[CSV EXPORT ERROR] " & ex.Message)
            Return False
        End Try
    End Function


    Public Shared Function ExportToPDF(dt As DataTable, documentTitle As String, filePath As String) As Boolean
        If dt Is Nothing OrElse dt.Rows.Count = 0 Then Return False

        Try
            Dim printDoc As New PrintDocument()
            printDoc.PrinterSettings.PrinterName = "Microsoft Print to PDF"
            printDoc.PrinterSettings.PrintToFile = True
            printDoc.PrinterSettings.PrintFileName = filePath

            Dim rowIndex As Integer = 0

            AddHandler printDoc.PrintPage, Sub(sender As Object, e As PrintPageEventArgs)
                                               Dim fontHeader As New Font("Arial", 14, FontStyle.Bold)
                                               Dim fontColumn As New Font("Arial", 10, FontStyle.Bold)
                                               Dim fontBody As New Font("Arial", 9, FontStyle.Regular)

                                               Dim x As Single = 40
                                               Dim y As Single = 40
                                               Dim rowHeight As Single = 25

                                               ' Title
                                               e.Graphics.DrawString(documentTitle, fontHeader, Brushes.Black, x, y)
                                               y += 40

                                               ' Table Headers
                                               Dim colWidth As Single = (e.MarginBounds.Width - 40) / dt.Columns.Count
                                               For c As Integer = 0 To dt.Columns.Count - 1
                                                   e.Graphics.DrawString(dt.Columns(c).ColumnName, fontColumn, Brushes.DarkBlue, x + (c * colWidth), y)
                                               Next
                                               y += rowHeight
                                               e.Graphics.DrawLine(Pens.Black, x, y, x + (dt.Columns.Count * colWidth), y)
                                               y += 5

                                               ' Data Rows
                                               While rowIndex < dt.Rows.Count
                                                   For c As Integer = 0 To dt.Columns.Count - 1
                                                       Dim cellVal As String = dt.Rows(rowIndex)(c).ToString()
                                                       e.Graphics.DrawString(cellVal, fontBody, Brushes.Black, x + (c * colWidth), y)
                                                   Next
                                                   y += rowHeight
                                                   rowIndex += 1

                                                   ' Check for Page Overflow
                                                   If y + rowHeight > e.MarginBounds.Bottom Then
                                                       e.HasMorePages = True
                                                       Return
                                                   End If
                                               End While

                                               e.HasMorePages = False
                                           End Sub

            printDoc.Print()
            Return True
        Catch ex As Exception
            Console.WriteLine("[PDF EXPORT ERROR] " & ex.Message)
            Return False
        End Try
    End Function

' ===== GradeCalculator.vb =====

' Pure grade math. No database, so it is easy to test.
Public Class GradeCalculator

    ' Passing mark. Change it here if your school uses a different one.
    Public Const PassingGrade As Double = 75.0

    Public Shared Function CalculateScorePercentage(score As Double, totalScore As Double) As Double
        If totalScore <= 0 Then Return 0.0
        Return Math.Round((score / totalScore) * 100, 2)
    End Function

    Public Shared Function DetermineRemarks(percentage As Double) As String
        Return If(percentage >= PassingGrade, "Passed", "Needs Improvement")
    End Function

    ' Weighted running grade over graded items only. Nothing graded yet = Nothing.
    Public Shared Function CalculateRunningGrade(items As IEnumerable(Of ScoreItem)) As Double?
        Dim earned As Double = 0.0
        Dim weightUsed As Double = 0.0

        For Each it As ScoreItem In items
            If it.Score.HasValue AndAlso it.TotalScore > 0 AndAlso it.Weight > 0 Then
                earned += (it.Score.Value / it.TotalScore) * it.Weight
                weightUsed += it.Weight
            End If
        Next

        If weightUsed = 0 Then Return Nothing
        Return Math.Round(earned / weightUsed * 100, 2)
    End Function

    Public Shared Function GetStatus(runningGrade As Double?) As String
        If Not runningGrade.HasValue Then Return "Not Yet"
        Return If(runningGrade.Value >= PassingGrade, "Passed", "At risk")
    End Function

End Class

    Public ReadOnly Property TotalScore As Double
    Public ReadOnly Property Weight As Double

    Public Sub New(score As Double?, totalScore As Double, weight As Double)
        Me.Score = score
        Me.TotalScore = totalScore
        Me.Weight = weight
    End Sub

' ===== GradeService.vb =====

Imports System.Data
Imports System.Globalization
Imports System.IO
Imports MySql.Data.MySqlClient

Public Class GradeService

    ' =========================================================
    '  RECORD SCORE (one transaction: score + running grade + audit row)
    ' =========================================================

    Public Function RecordScore(enrollmentId As Integer, assessmentId As Integer,
                                newScore As Double, teacherId As String,
                                Optional reason As String = Nothing) As OperationResult
        Try
            Using conn As MySqlConnection = DatabaseHelper.GetConnection()
                Using tx As MySqlTransaction = conn.BeginTransaction()
                    Try
                        Dim res As OperationResult = RecordScoreTx(conn, tx, enrollmentId, assessmentId, newScore, teacherId, reason)
                        If res.Success Then tx.Commit() Else tx.Rollback()
                        Return res
                    Catch
                        tx.Rollback()
                        Throw
                    End Try
                End Using
            End Using
        Catch ex As Exception
            Console.WriteLine("[RECORD SCORE ERROR] " & ex.Message)
            Return OperationResult.Fail("Could not save the score: " & ex.Message)
        End Try
    End Function

    ' Same logic inside a transaction somebody else owns (the clarification approval will reuse this).
    ' It never commits or rolls back by itself.
    Public Function RecordScoreTx(conn As MySqlConnection, tx As MySqlTransaction,
                                  enrollmentId As Integer, assessmentId As Integer,
                                  newScore As Double, teacherId As String,
                                  Optional reason As String = Nothing) As OperationResult

        ' 1) Who owns this class, and how many points is the assessment worth?
        Dim totalScore As Double = 0
        Dim title As String = ""
        Dim ownerId As String = ""
        Dim sameClass As Boolean = False

        Dim infoSql As String =
            "SELECT a.total_score, a.title, c.teacher_id, (a.class_id = e.class_id) AS same_class " &
            "FROM enrollment e " &
            "JOIN class c ON e.class_id = c.class_id " &
            "JOIN assessment a ON a.assessment_id = @AssessmentId " &
            "WHERE e.enrollment_id = @EnrollmentId"

        Using cmd As New MySqlCommand(infoSql, conn, tx)
            cmd.Parameters.AddWithValue("@AssessmentId", assessmentId)
            cmd.Parameters.AddWithValue("@EnrollmentId", enrollmentId)
            Using r As MySqlDataReader = cmd.ExecuteReader()
                If Not r.Read() Then Return OperationResult.Fail("Enrollment or assessment not found.")
                totalScore = Convert.ToDouble(r.GetValue(0))
                title = r.GetValue(1).ToString()
                ownerId = r.GetValue(2).ToString()
                sameClass = (Convert.ToInt32(r.GetValue(3)) = 1)
            End Using
        End Using

        If Not String.Equals(ownerId, teacherId, StringComparison.OrdinalIgnoreCase) Then
            Return OperationResult.Fail("You can only record scores for your own classes.")
        End If
        If Not sameClass Then
            Return OperationResult.Fail("That assessment belongs to a different class.")
        End If
        If Not ValidationHelper.IsValidScore(newScore, totalScore) Then
            Return OperationResult.Fail("Score must be between 0 and " & Fmt(totalScore) & ".")
        End If

        ' 2) Lock the current row and read the old score (NULL = not graded yet)
        Dim oldScore As Double? = Nothing
        Dim lockSql As String = "SELECT score FROM studentgrade WHERE enrollment_id = @EnrollmentId AND assessment_id = @AssessmentId FOR UPDATE"
        Using cmd As New MySqlCommand(lockSql, conn, tx)
            cmd.Parameters.AddWithValue("@EnrollmentId", enrollmentId)
            cmd.Parameters.AddWithValue("@AssessmentId", assessmentId)
            oldScore = ToNullableDouble(cmd.ExecuteScalar())
        End Using

        If oldScore.HasValue AndAlso oldScore.Value = newScore Then
            Return OperationResult.Ok("Score unchanged.")
        End If

        ' 3) Save the score (the row usually exists already; insert if it does not)
        Dim upsertSql As String =
            "INSERT INTO studentgrade (enrollment_id, assessment_id, score, remarks) " &
            "VALUES (@EnrollmentId, @AssessmentId, @Score, @Remarks) " &
            "ON DUPLICATE KEY UPDATE score = VALUES(score), remarks = VALUES(remarks)"
        Using cmd As New MySqlCommand(upsertSql, conn, tx)
            cmd.Parameters.AddWithValue("@EnrollmentId", enrollmentId)
            cmd.Parameters.AddWithValue("@AssessmentId", assessmentId)
            cmd.Parameters.AddWithValue("@Score", newScore)
            cmd.Parameters.AddWithValue("@Remarks", GradeCalculator.DetermineRemarks(GradeCalculator.CalculateScorePercentage(newScore, totalScore)))
            cmd.ExecuteNonQuery()
        End Using

        ' 4) Refresh the cached running grade on the enrollment
        RefreshRunningGrade(conn, tx, enrollmentId)

        ' 5) Audit row (same transaction, so it rolls back with the score)
        Dim gradeId As String
        Using cmd As New MySqlCommand("SELECT grade_id FROM studentgrade WHERE enrollment_id = @EnrollmentId AND assessment_id = @AssessmentId", conn, tx)
            cmd.Parameters.AddWithValue("@EnrollmentId", enrollmentId)
            cmd.Parameters.AddWithValue("@AssessmentId", assessmentId)
            gradeId = cmd.ExecuteScalar().ToString()
        End Using

        Dim oldText As String = Nothing
        If oldScore.HasValue Then oldText = Fmt(oldScore.Value) & "/" & Fmt(totalScore)

        AuditService.LogActionTx(conn, tx, "Teacher", teacherId,
                                 If(oldScore.HasValue, "Updated score", "Recorded score"),
                                 "StudentGrade", gradeId,
                                 oldText, Fmt(newScore) & "/" & Fmt(totalScore), reason)

        Return OperationResult.Ok("Saved " & title & ": " & Fmt(newScore) & "/" & Fmt(totalScore))
    End Function

    Private Shared Function RefreshRunningGrade(conn As MySqlConnection, tx As MySqlTransaction, enrollmentId As Integer) As Double?
        Dim items As New List(Of ScoreItem)()
        Dim sql As String =
            "SELECT sg.score, a.total_score, a.weight " &
            "FROM studentgrade sg JOIN assessment a ON sg.assessment_id = a.assessment_id " &
            "WHERE sg.enrollment_id = @EnrollmentId"

        Using cmd As New MySqlCommand(sql, conn, tx)
            cmd.Parameters.AddWithValue("@EnrollmentId", enrollmentId)
            Using r As MySqlDataReader = cmd.ExecuteReader()
                While r.Read()
                    items.Add(New ScoreItem(ToNullableDouble(r.GetValue(0)),
                                            Convert.ToDouble(r.GetValue(1)),
                                            Convert.ToDouble(r.GetValue(2))))
                End While
            End Using
        End Using

        Dim running As Double? = GradeCalculator.CalculateRunningGrade(items)

        Using cmd As New MySqlCommand("UPDATE enrollment SET running_grade = @Running WHERE enrollment_id = @EnrollmentId", conn, tx)
            If running.HasValue Then
                cmd.Parameters.AddWithValue("@Running", running.Value)
            Else
                cmd.Parameters.AddWithValue("@Running", DBNull.Value)
            End If
            cmd.Parameters.AddWithValue("@EnrollmentId", enrollmentId)
            cmd.ExecuteNonQuery()
        End Using

        Return running
    End Function

    ' One-time helper: fills running_grade for every enrollment (the seed data has them all NULL)
    Public Function RecalculateAllRunningGrades() As Integer
        Using conn As MySqlConnection = DatabaseHelper.GetConnection()
            Using tx As MySqlTransaction = conn.BeginTransaction()
                Dim ids As New List(Of Integer)()
                Using cmd As New MySqlCommand("SELECT enrollment_id FROM enrollment", conn, tx)
                    Using r As MySqlDataReader = cmd.ExecuteReader()
                        While r.Read()
                            ids.Add(Convert.ToInt32(r.GetValue(0)))
                        End While
                    End Using
                End Using

                For Each id As Integer In ids
                    RefreshRunningGrade(conn, tx, id)
                Next

                tx.Commit()
                Return ids.Count
            End Using
        End Using
    End Function

            "WHERE sg.enrollment_id = @EnrollmentId AND e.student_id = @StudentId " &
            "ORDER BY a.term, a.created_at, a.assessment_id"
        Return DatabaseHelper.ExecuteQuery(sql, {New MySqlParameter("@EnrollmentId", enrollmentId), New MySqlParameter("@StudentId", studentId)})
    End Function

    ' Grade History screen: reads score changes from auditlog
    Public Function GetGradeHistoryForStudent(studentId As String) As DataTable
        Dim sql As String =
            "SELECT al.timestamp AS changed_at, a.title AS assessment, al.old_value, al.new_value, " &
            "       COALESCE(t.full_name, al.teacher_id) AS changed_by, al.details AS reason " &
            "FROM auditlog al " &
            "JOIN studentgrade sg ON sg.grade_id = CAST(al.entity_id AS UNSIGNED) " &
            "JOIN enrollment e ON sg.enrollment_id = e.enrollment_id " &
            "JOIN assessment a ON sg.assessment_id = a.assessment_id " &
            "LEFT JOIN teacher t ON al.teacher_id = t.teacher_id " &
            "WHERE al.entity_type = 'StudentGrade' AND e.student_id = @StudentId " &
            "ORDER BY al.timestamp DESC, al.log_id DESC"
        Return DatabaseHelper.ExecuteQuery(sql, {New MySqlParameter("@StudentId", studentId)})
    End Function

    ' Class Grades table: one row per student, one column per assessment.
    ' term = "All", "Midterm" or "Finals" ("Midterms" is accepted too).
    ' Assessment columns are named "A<assessment_id>" and the title is in the column Caption.
    Public Function GetClassGradeSheet(classId As Integer, teacherId As String,
                                       Optional term As String = "All") As DataTable
        If term = "Midterms" Then term = "Midterm"
        Dim filterTerm As Boolean = Not String.IsNullOrEmpty(term) AndAlso
                                    Not term.Equals("All", StringComparison.OrdinalIgnoreCase)

        ' Assessment columns (empty if this teacher does not own the class)
        Dim aSql As String =
            "SELECT a.assessment_id, a.title FROM assessment a " &
            "JOIN class c ON a.class_id = c.class_id " &
            "WHERE a.class_id = @ClassId AND c.teacher_id = @TeacherId"
        Dim aParams As New List(Of MySqlParameter) From {
            New MySqlParameter("@ClassId", classId),
            New MySqlParameter("@TeacherId", teacherId)}
        If filterTerm Then
            aSql &= " AND a.term = @Term"
            aParams.Add(New MySqlParameter("@Term", term))
        End If
        aSql &= " ORDER BY a.term, a.created_at, a.assessment_id"
        Dim assessments As DataTable = DatabaseHelper.ExecuteQuery(aSql, aParams.ToArray())

        Dim students As DataTable = GetEnrollmentsByClass(classId, teacherId)

        Dim gradesSql As String =
            "SELECT sg.enrollment_id, sg.assessment_id, sg.score, a.total_score " &
            "FROM studentgrade sg " &
            "JOIN assessment a ON sg.assessment_id = a.assessment_id " &
            "JOIN enrollment e ON sg.enrollment_id = e.enrollment_id " &
            "JOIN class c ON e.class_id = c.class_id " &
            "WHERE e.class_id = @ClassId AND c.teacher_id = @TeacherId"
        Dim grades As DataTable = DatabaseHelper.ExecuteQuery(gradesSql,
            {New MySqlParameter("@ClassId", classId), New MySqlParameter("@TeacherId", teacherId)})

        Dim cells As New Dictionary(Of String, String)()
        For Each g As DataRow In grades.Rows
            Dim key As String = g("enrollment_id").ToString() & ":" & g("assessment_id").ToString()
            If Convert.IsDBNull(g("score")) Then
                cells(key) = "-"
            Else
                cells(key) = Fmt(Convert.ToDouble(g("score"))) & "/" & Fmt(Convert.ToDouble(g("total_score")))
            End If
        Next

        Dim sheet As New DataTable()
        sheet.Columns.Add("EnrollmentId", GetType(Integer))
        sheet.Columns.Add("StudentId", GetType(String))
        sheet.Columns.Add("Student", GetType(String))
        For Each a As DataRow In assessments.Rows
            Dim col As DataColumn = sheet.Columns.Add("A" & a("assessment_id").ToString(), GetType(String))
            col.Caption = a("title").ToString()
        Next
        sheet.Columns.Add("RunningGrade", GetType(Double))
        sheet.Columns.Add("Status", GetType(String))

        For Each s As DataRow In students.Rows
            Dim row As DataRow = sheet.NewRow()
            row("EnrollmentId") = Convert.ToInt32(s("enrollment_id"))
            row("StudentId") = s("student_id").ToString()
            row("Student") = s("full_name").ToString()

            For Each a As DataRow In assessments.Rows
                Dim cell As String = Nothing
                If Not cells.TryGetValue(s("enrollment_id").ToString() & ":" & a("assessment_id").ToString(), cell) Then
                    cell = "-"
                End If
                row("A" & a("assessment_id").ToString()) = cell
            Next

            Dim rg As Double? = ToNullableDouble(s("running_grade"))
            If rg.HasValue Then
                row("RunningGrade") = rg.Value
            Else
                row("RunningGrade") = DBNull.Value
            End If
            row("Status") = GradeCalculator.GetStatus(rg)

            sheet.Rows.Add(row)
        Next

        Return sheet
    End Function

    ' =========================================================
    '  CSV IMPORT (reuses RecordScore, so every row is checked and audited)
    ' =========================================================

    Public Function BulkImportGradesFromCSV(classId As Integer, assessmentId As Integer,
                                            filePath As String, teacherId As String) As (SuccessCount As Integer, FailCount As Integer)
        Dim successCount As Integer = 0
        Dim failCount As Integer = 0

        If Not File.Exists(filePath) Then Return (0, 0)

        Dim lines() As String = File.ReadAllLines(filePath)

        ' Row 0 is the header
        For i As Integer = 1 To lines.Length - 1
            Dim line As String = lines(i).Trim()
            If String.IsNullOrWhiteSpace(line) Then Continue For

            Dim parts() As String = line.Split(","c)
            If parts.Length < 2 Then
                failCount += 1
                Continue For
            End If

            Dim studentId As String = parts(0).Trim().Replace("""", "")
            Dim scoreVal As Double
            If Not Double.TryParse(parts(1).Trim().Replace("""", ""), NumberStyles.Float, CultureInfo.InvariantCulture, scoreVal) Then
                failCount += 1
                Continue For
            End If

            Dim enrollmentId As Integer = GetEnrollmentIdByStudentAndClass(studentId, classId)
            If enrollmentId <= 0 Then
                failCount += 1
                Continue For
            End If

            If RecordScore(enrollmentId, assessmentId, scoreVal, teacherId, "Bulk CSV Upload").Success Then
                successCount += 1
            Else
                failCount += 1
            End If
        Next

        Return (successCount, failCount)
    End Function

    Private Function GetEnrollmentIdByStudentAndClass(studentId As String, classId As Integer) As Integer
        Dim sql As String = "SELECT enrollment_id FROM enrollment WHERE student_id = @StudentId AND class_id = @ClassId"
        Dim result As Object = DatabaseHelper.ExecuteScalar(sql,
            {New MySqlParameter("@StudentId", studentId), New MySqlParameter("@ClassId", classId)})
        If result IsNot Nothing AndAlso Not Convert.IsDBNull(result) Then
            Return Convert.ToInt32(result)
        End If
        Return 0
    End Function

    ' =========================================================
    '  SMALL HELPERS
    ' =========================================================

    Private Shared Function ToNullableDouble(o As Object) As Double?
        If o Is Nothing OrElse Convert.IsDBNull(o) Then Return Nothing
        Return Convert.ToDouble(o)
    End Function

    ' 18 -> "18", 92.5 -> "92.5"
    Private Shared Function Fmt(value As Double) As String
        Return value.ToString("0.##", CultureInfo.InvariantCulture)
    End Function

End Class

Public Class RevisionRequestService

    Public Function SubmitRequest(request As RevisionRequest, studentId As String) As Boolean
        Dim sql As String = "INSERT INTO revisionrequest (grade_id, requested_points, reason, status, requested_at) " &
                            "VALUES (@GradeId, @RequestedPoints, @Reason, 'Pending', NOW())"

        Dim params As MySqlParameter() = {
            New MySqlParameter("@GradeId", request.GradeId),
            New MySqlParameter("@RequestedPoints", If(request.RequestedPoints.HasValue, request.RequestedPoints.Value, DBNull.Value)),
            New MySqlParameter("@Reason", ValidationHelper.SanitizeInput(request.Reason))
        }

        Try
            Dim rowsAffected As Integer = DatabaseHelper.ExecuteNonQuery(sql, params)
            If rowsAffected > 0 Then
                AuditService.LogAction("Student", studentId, "SUBMIT_CLARIFICATION_REQUEST", "studentgrade", request.GradeId.ToString(), details:=request.Reason)
                Return True
            End If
            Return False
        Catch ex As Exception
            Console.WriteLine("[REVISION SERVICE ERROR] " & ex.Message)
            Return False
        End Try
    End Function

    Public Function ReviewRequest(requestId As Integer, status As String, teacherId As String) As Boolean
        Dim sql As String = "UPDATE revisionrequest " &
                            "SET status = @Status, reviewed_at = NOW(), reviewed_by = @TeacherId " &
                            "WHERE request_id = @RequestId"

        Dim params As MySqlParameter() = {
            New MySqlParameter("@Status", status),
            New MySqlParameter("@TeacherId", teacherId),
            New MySqlParameter("@RequestId", requestId)
        }

        Try
            Dim rowsAffected As Integer = DatabaseHelper.ExecuteNonQuery(sql, params)
            If rowsAffected > 0 Then
                AuditService.LogAction("Teacher", teacherId, $"REVISION_{status.ToUpper()}", "revisionrequest", requestId.ToString(), details:=$"Status set to {status}")
                Return True
            End If
            Return False
        Catch ex As Exception
            Console.WriteLine("[REVISION SERVICE ERROR] " & ex.Message)
            Return False
        End Try
    End Function

    Public Function GetPendingRequestsByTeacher(teacherId As String) As DataTable
        Dim sql As String = "SELECT r.request_id, r.grade_id, s.full_name AS student_name, a.title AS assessment_title, " &
                            "sg.score AS current_score, r.requested_points, r.reason, r.status, r.requested_at " &
                            "FROM revisionrequest r " &
                            "JOIN studentgrade sg ON r.grade_id = sg.grade_id " &
                            "JOIN enrollment e ON sg.enrollment_id = e.enrollment_id " &
                            "JOIN student s ON e.student_id = s.student_id " &
                            "JOIN class c ON e.class_id = c.class_id " &
                            "JOIN assessment a ON sg.assessment_id = a.assessment_id " &
                            "WHERE c.teacher_id = @TeacherId AND r.status = 'Pending' " &
                            "ORDER BY r.requested_at DESC"

        Dim params As MySqlParameter() = {
            New MySqlParameter("@TeacherId", teacherId)
        }

        Return DatabaseHelper.ExecuteQuery(sql, params)
    End Function
