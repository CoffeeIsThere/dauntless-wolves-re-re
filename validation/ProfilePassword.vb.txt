Imports MySql.Data.MySqlClient
Imports DAUNTLESS_WOLVES_PROF.Models

' Mutates a disposable seeded database. Never use against actual school records.
Module ProfilePasswordChecks
    Private checks As Integer
    Sub Main()
        Const student As String = "25-05764"
        Const teacher As String = "P001"
        Dim service As New StudentProfileService()
        AppSession.ClearSession()
        Dim denied = False
        Try
            service.LoadCurrent()
        Catch ex As UnauthorizedAccessException
            denied = True
        End Try
        Check(denied, "profile rejects missing session")
        AppSession.StartSession(New UserModel With {.Role = "Teacher", .UserId = teacher})
        denied = False
        Try
            service.LoadCurrent()
        Catch ex As UnauthorizedAccessException
            denied = True
        End Try
        Check(denied, "profile rejects nonstudent session")
        Dim hash = SecurityHelper.HashPassword("old-test-password")
        DatabaseHelper.ExecuteNonQuery("UPDATE teacher SET password_hash=@Hash WHERE teacher_id='P001'", New MySqlParameter("@Hash", hash))
        DatabaseHelper.ExecuteNonQuery("INSERT INTO student(student_id,full_name,email,password_hash,program,year_level,section,status) VALUES('99-00001','Empty Profile','empty-profile@test.invalid',@Hash,'Testing',1,'T','Active')", New MySqlParameter("@Hash", hash))
        AppSession.StartSession(New UserModel With {.Role = "Student", .UserId = "99-00001"})
        Dim empty = service.LoadCurrent()
        Check(empty.Classes.Rows.Count = 0 AndAlso empty.OpenRequests = 0, "student without enrollments has empty summaries")
        Check(empty.Profile("student_id").ToString() = "99-00001", "profile belongs to current student")

        DatabaseHelper.ExecuteNonQuery("INSERT INTO course(course_code,course_name) VALUES('PROFILETEST','Profile Test')")
        DatabaseHelper.ExecuteNonQuery("INSERT INTO class(course_code,section,semester,academic_year,teacher_id) VALUES('PROFILETEST','A','1st Semester','2026-2027','P001'),('PROFILETEST','A','2nd Semester','2026-2027','P001')")
        Dim classId = CInt(DatabaseHelper.ExecuteScalar("SELECT MIN(class_id) FROM class WHERE course_code='PROFILETEST'"))
        Dim ungradedClass = CInt(DatabaseHelper.ExecuteScalar("SELECT MAX(class_id) FROM class WHERE course_code='PROFILETEST'"))
        For Each id In {classId, ungradedClass}
            DatabaseHelper.ExecuteNonQuery("INSERT INTO enrollment(student_id,class_id) VALUES(@Student,@Class)", New MySqlParameter("@Student", student), New MySqlParameter("@Class", id))
        Next
        Dim enrollment = CInt(DatabaseHelper.ExecuteScalar("SELECT enrollment_id FROM enrollment WHERE student_id=@Student AND class_id=@Class", New MySqlParameter("@Student", student), New MySqlParameter("@Class", classId)))
        For Each item In {("Zero", 20, 10), ("Weighted", 100, 30), ("Pending", 10, 60)}
            DatabaseHelper.ExecuteNonQuery("INSERT INTO assessment(class_id,term,title,assessment_type,total_score,weight) VALUES(@Class,'Midterm',@Title,'Quiz',@Max,@Weight)", New MySqlParameter("@Class", classId), New MySqlParameter("@Title", item.Item1), New MySqlParameter("@Max", item.Item2), New MySqlParameter("@Weight", item.Item3))
        Next
        DatabaseHelper.ExecuteNonQuery("INSERT INTO studentgrade(enrollment_id,assessment_id,score) SELECT @Enrollment,assessment_id,CASE title WHEN 'Zero' THEN 0 WHEN 'Weighted' THEN 90 ELSE NULL END FROM assessment WHERE class_id=@Class", New MySqlParameter("@Enrollment", enrollment), New MySqlParameter("@Class", classId))
        DatabaseHelper.ExecuteNonQuery("DELETE r FROM revisionrequest r JOIN studentgrade sg ON sg.grade_id=r.grade_id JOIN enrollment e ON e.enrollment_id=sg.enrollment_id WHERE e.student_id=@Student", New MySqlParameter("@Student", student))
        DatabaseHelper.ExecuteNonQuery("INSERT INTO revisionrequest(grade_id,reason,status) SELECT sg.grade_id,'Profile check',CASE a.title WHEN 'Zero' THEN 'Pending' ELSE 'Under Review' END FROM studentgrade sg JOIN assessment a ON a.assessment_id=sg.assessment_id WHERE sg.enrollment_id=@Enrollment AND sg.score IS NOT NULL", New MySqlParameter("@Enrollment", enrollment))
        AppSession.StartSession(New UserModel With {.Role = "Student", .UserId = student})
        Dim snapshot = service.LoadCurrent()
        Dim row = snapshot.Classes.AsEnumerable().Single(Function(item) item("Class").ToString().StartsWith("PROFILETEST") AndAlso item("Class").ToString().Contains("1st Semester"))
        Check(CDbl(row("Running Grade (%)")) = 67.5, "weighted grade includes zero and excludes missing scores")
        Check(row("Standing").ToString() = "At risk", "standing follows shared threshold")
        Dim pending = snapshot.Classes.AsEnumerable().Single(Function(item) item("Class").ToString().StartsWith("PROFILETEST") AndAlso item("Class").ToString().Contains("2nd Semester"))
        Check(IsDBNull(pending("Running Grade (%)")) AndAlso pending("Standing").ToString() = "Not Yet", "class without assessments is ungraded rather than zero")
        Check(snapshot.OpenRequests = 2, "open count includes Pending and Under Review")
        Check(snapshot.Classes.Rows.Count = CInt(DatabaseHelper.ExecuteScalar("SELECT COUNT(*) FROM enrollment WHERE student_id=@Student", New MySqlParameter("@Student", student))), "class count matches own enrollments")
        Check(row("Class").ToString().Contains("2026-2027"), "class labels distinguish academic offerings")
        DatabaseHelper.ExecuteNonQuery("UPDATE revisionrequest r JOIN studentgrade sg ON sg.grade_id=r.grade_id SET r.status='Closed' WHERE sg.enrollment_id=@Enrollment", New MySqlParameter("@Enrollment", enrollment))
        Check(service.LoadCurrent().OpenRequests = 0, "fresh snapshot excludes closed requests")
        DatabaseHelper.ExecuteNonQuery("UPDATE student SET program='Updated program' WHERE student_id=@Student", New MySqlParameter("@Student", student))
        Check(service.LoadCurrent().Profile("program").ToString() = "Updated program", "refresh reads updated profile")
        AppSession.StartSession(New UserModel With {.Role = "Student", .UserId = "99-00001"})
        Check(service.LoadCurrent().Classes.Rows.Count = 0, "another student cannot see previous student's classes")
        Check(Not service.LoadCurrent().Profile.Table.Columns.Contains("password_hash"), "profile never exposes password hashes")

        AppSession.StartSession(New UserModel With {.Role = "Teacher", .UserId = teacher})
        Dim auth As New AuthService()
        Dim auditBefore = CInt(DatabaseHelper.ExecuteScalar("SELECT COUNT(*) FROM auditlog WHERE teacher_id='P001' AND action='Changed password'"))
        Check(Not auth.ChangePassword("Teacher", teacher, "incorrect", "new-test-password").Success, "incorrect current teacher password rejected")
        Check(SecurityHelper.VerifyPassword("old-test-password", CStr(DatabaseHelper.ExecuteScalar("SELECT password_hash FROM teacher WHERE teacher_id='P001'"))), "failed password change preserves old password")
        Check(auth.ChangePassword("Teacher", teacher, "old-test-password", "new-test-password").Success, "teacher can change own password")
        Check(auth.VerifyLogin(teacher, "new-test-password", "Teacher") IsNot Nothing, "new teacher password authenticates")
        Check(auth.VerifyLogin(teacher, "old-test-password", "Teacher") Is Nothing, "old teacher password no longer authenticates")
        Check(CInt(DatabaseHelper.ExecuteScalar("SELECT COUNT(*) FROM auditlog WHERE teacher_id='P001' AND action='Changed password'")) = auditBefore + 1, "one teacher password audit entry committed")
        DatabaseHelper.ExecuteNonQuery("UPDATE teacher SET status='Inactive' WHERE teacher_id='P001'")
        Check(Not auth.ChangePassword("Teacher", teacher, "new-test-password", "another-test-password").Success, "inactive teacher cannot change password")
        Console.WriteLine(checks.ToString() & " checks passed.")
    End Sub

    Private Sub Check(ok As Boolean, description As String)
        If Not ok Then Throw New Exception("FAIL: " & description)
        checks += 1
        Console.WriteLine("PASS: " & description)
    End Sub
End Module
