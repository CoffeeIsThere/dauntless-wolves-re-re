Imports MySql.Data.MySqlClient
Public Module AppSession
    Public IsAdmin As Boolean = False
    Public IsTeacher As Boolean = True
    Public CurrentUser As New User()
End Module
Public Class User
    Public UserId As String
End Class
Module Program
    Sub Main()
        Dim maximum, weight As Double
        Check(ValidationHelper.TryAssessmentImportRow({"Quiz", "Quiz", "20", "Midterm", "10"}, maximum, weight), "valid assessment import row")
        Check(Not ValidationHelper.TryAssessmentImportRow({"Quiz", "Quiz", "NaN", "Midterm", "10"}, maximum, weight), "reject nonfinite import score")
        Check(Not ValidationHelper.TryAssessmentImportRow({"Quiz", "Quiz", "20", "1st Semester", "10"}, maximum, weight), "reject import semester")
        Check(Not ValidationHelper.TryAssessmentImportRow({"Quiz", "Quiz", "20", "Finals", "101"}, maximum, weight), "reject invalid import weight")
        Check(Not ValidationHelper.TryAssessmentImportRow({"Unknown", "CW", "20", "Finals", "10"}, maximum, weight), "reject unsupported import type")
        Dim service As New GradeService()
        Using conn = DatabaseHelper.GetConnection()
            Dim teacher As String
            Dim classId As Integer
            Using cmd As New MySqlCommand("SELECT teacher_id,class_id FROM class ORDER BY class_id LIMIT 1", conn)
                Using r = cmd.ExecuteReader()
                    r.Read()
                    teacher = r.GetString(0)
                    classId = r.GetInt32(1)
                End Using
            End Using
            Check(Not service.CreateAssessment(classId, "Denied", "Quiz", 20, "Midterm", "not-owner").Success, "reject other teacher")
            Check(Not service.CreateAssessment(classId, "Invalid", "Quiz", 20, "1st Semester", teacher).Success, "reject semester as term")
            Check(service.CreateAssessment(classId, "Review test", "Quiz", 20, "Midterm", teacher, 10).Success, "create assessment")
            Dim id As Integer
            Using cmd As New MySqlCommand("SELECT MAX(assessment_id) FROM assessment", conn)
                id = CInt(cmd.ExecuteScalar())
            End Using
            Dim student As String
            Using cmd As New MySqlCommand("SELECT student_id FROM enrollment WHERE class_id=" & classId & " LIMIT 1", conn)
                student = CStr(cmd.ExecuteScalar())
            End Using
            Check(service.SaveAssessmentScore(student, classId, id, 15, teacher).Success, "save score")
            Check(Not service.SaveAssessmentScore(student, classId, id, 25, teacher).Success, "reject out of range")
            Check(Not service.SaveAssessmentScore(student, classId, id, Nothing, teacher).Success, "reject silent clearing")
            Using cmd As New MySqlCommand("SELECT score FROM studentgrade WHERE assessment_id=" & id & " AND enrollment_id=(SELECT enrollment_id FROM enrollment WHERE class_id=" & classId & " AND student_id=@Student)", conn)
                cmd.Parameters.AddWithValue("@Student", student)
                Check(CDbl(cmd.ExecuteScalar()) = 15, "saved score survives rejected edits")
            End Using
            Check(service.GetClassGradeSheet(classId, teacher).Columns.Contains("A" & id), "grade sheet uses assessment ID")
            Check(service.GetAuditLogsForTeacher(teacher).Rows.Count > 0, "audit query returns changes")
            Dim revision As New RevisionRequestService()
            Dim requestId As Integer
            Using cmd As New MySqlCommand("INSERT INTO revisionrequest (grade_id, reason, requested_points, status) SELECT grade_id, 'Integration check', 2, 'Pending' FROM studentgrade WHERE assessment_id=@Id AND score=15 LIMIT 1; SELECT LAST_INSERT_ID();", conn)
                cmd.Parameters.AddWithValue("@Id", id)
                requestId = CInt(cmd.ExecuteScalar())
            End Using
            Check(Not revision.ApproveAndApplyRevision(requestId, "not-owner").Success, "deny approval by other teacher")
            Check(revision.ReviewRequestResult(requestId, "Under Review", teacher).Success, "mark under review")
            Check(revision.ApproveAndApplyRevision(requestId, teacher).Success, "approve under-review points request")
            Check(Not revision.ApproveAndApplyRevision(requestId, teacher).Success, "prevent duplicate approval")
            Using cmd As New MySqlCommand("INSERT INTO revisionrequest (grade_id, reason, status) SELECT grade_id, 'Recheck test', 'Under Review' FROM studentgrade WHERE assessment_id=@Id AND score=17 LIMIT 1; SELECT LAST_INSERT_ID();", conn)
                cmd.Parameters.AddWithValue("@Id", id)
                requestId = CInt(cmd.ExecuteScalar())
            End Using
            Check(revision.ApproveAndApplyRevision(requestId, teacher).Message = "NEEDSCORE", "recheck requires corrected score")
            Check(Not revision.ApproveAndApplyRevision(requestId, teacher, 25).Success, "reject invalid corrected score")
            Check(revision.ApproveAndApplyRevision(requestId, teacher, 18, "Rechecked your submitted answer.").Success, "approve corrected recheck score")
            Check(revision.GetRevisionRequestsByStudent(student).Select("request_id=" & requestId)(0)("reviewer_response").ToString().Contains("Rechecked your submitted answer."), "student sees approval response")
            Using cmd As New MySqlCommand("INSERT INTO revisionrequest (grade_id, reason, status) SELECT grade_id, 'Close test', 'Pending' FROM studentgrade WHERE assessment_id=@Id AND score=18 LIMIT 1; SELECT LAST_INSERT_ID();", conn)
                cmd.Parameters.AddWithValue("@Id", id)
                requestId = CInt(cmd.ExecuteScalar())
            End Using
            Check(revision.RejectRequestResult(requestId, teacher, "Insufficient evidence").Success, "close request with reason")
            Check(revision.GetRevisionRequestsByStudent(student).Select("request_id=" & requestId)(0)("reviewer_response").ToString() = "Insufficient evidence", "student sees closure reason")
            Check(revision.GetRevisionRequestsByStudent("not-this-student").Rows.Count = 0, "other student cannot read responses")
            Dim browserRequests = revision.GetRequestsByTeacher(teacher, "All")
            Check(browserRequests.Select("request_id=" & requestId).Length = 1, "request chooser includes closed decisions")
            Check(CInt(browserRequests.Select("request_id=" & requestId)(0)("class_id")) = classId, "request chooser supplies correct class identity")
            Check(revision.GetRequestsByTeacher("not-this-teacher", "All").Rows.Count = 0, "request chooser excludes other professors' records")
            Check(revision.GetRequestsByTeacher(teacher, "Open").Select("request_id=" & requestId).Length = 0, "open filter excludes closed requests")
            AppSession.CurrentUser.UserId = teacher
            Dim registeredName As String
            Using cmd As New MySqlCommand("SELECT full_name FROM student WHERE student_id=@Id", conn)
                cmd.Parameters.AddWithValue("@Id", student)
                registeredName = CStr(cmd.ExecuteScalar())
            End Using
            Dim importer As New ClassImportService()
            Dim importedClass As Integer
            Dim code = "TEST" & Guid.NewGuid().ToString("N").Substring(0, 8)
            Dim roster As New List(Of String()) From {
                New String() {"student_id", "full_name", "Quiz 1 /20", "Quiz 2 /20", "Quiz 3 /30"},
                New String() {student, registeredName, "18", "", "25"}}
            Check(importer.ImportClass(code, "Import test", "A", "1st Semester", "2026-2027", "Midterm", roster, importedClass).Success, "import class roster with three quizzes")
            Using cmd As New MySqlCommand("SELECT COUNT(*) FROM assessment WHERE class_id=" & importedClass, conn)
                Check(CInt(cmd.ExecuteScalar()) = 3, "all CSV quiz headers create assessments")
            End Using
            Using cmd As New MySqlCommand("SELECT COUNT(*) FROM studentgrade sg JOIN enrollment e ON e.enrollment_id=sg.enrollment_id WHERE e.class_id=" & importedClass & " AND sg.score IS NULL", conn)
                Check(CInt(cmd.ExecuteScalar()) = 1, "blank imported score remains ungraded")
            End Using
            Check(service.GetClassGradeSheet(importedClass, teacher).Rows.Count = 1, "imported roster appears in grade sheet")
            roster(1)(0) = "UNKNOWN"
            Dim failedCode = "FAIL" & Guid.NewGuid().ToString("N").Substring(0, 8)
            Check(Not importer.ImportClass(failedCode, "Rollback test", "A", "1st Semester", "2026-2027", "Midterm", roster, importedClass).Success, "reject unregistered student")
            Using cmd As New MySqlCommand("SELECT COUNT(*) FROM course WHERE course_code=@Code", conn)
                cmd.Parameters.AddWithValue("@Code", failedCode)
                Check(CInt(cmd.ExecuteScalar()) = 0, "failed roster import rolls back course and class")
            End Using
            roster(1)(0) = student
            roster(1)(2) = "21"
            Check(Not importer.ImportClass("BAD", "Bad score", "A", "1st Semester", "2026-2027", "Midterm", roster, importedClass).Success, "reject imported score above maximum")
            Check(GradeCalculator.CalculateRunningGrade({New ScoreItem(14.99, 20, 10)}).Value = 74.95, "standing uses full precision below passing boundary")
            Check(GradeCalculator.CalculateRunningGrade({New ScoreItem(0, 20, 10)}).Value = 0, "zero grade remains a real grade")
            Check(GradeCalculator.CalculateRunningGrade({New ScoreItem(10, 20, 10), New ScoreItem(100, 100, 90)}).Value = 95, "weighted average differs from simple average")
            Dim historyCount = service.GetGradeHistoryForStudent(student).Rows.Count
            Dim gradeId = CInt(DatabaseHelper.ExecuteScalar("SELECT sg.grade_id FROM studentgrade sg JOIN enrollment e ON e.enrollment_id=sg.enrollment_id WHERE sg.assessment_id=@Id AND e.student_id=@Student", New MySqlParameter("@Id", id), New MySqlParameter("@Student", student)))
            AuditService.LogAction("Teacher", teacher, "Unrelated enrollment event", "Enrollment", gradeId.ToString())
            Check(service.GetGradeHistoryForStudent(student).Rows.Count = historyCount, "history excludes numeric ID collisions with other entity types")
            DatabaseHelper.ExecuteNonQuery("UPDATE teacher SET status='Inactive' WHERE teacher_id=@Id", New MySqlParameter("@Id", teacher))
            Try
                Check(Not New ClassService().EnrollStudent(student, classId, "Teacher", teacher).Success, "deactivated professor cannot enroll students")
                Check(Not service.SaveAssessmentScore(student, classId, id, 19, teacher).Success, "deactivated professor cannot save grades in existing session")
                Check(Not service.CreateAssessment(classId, "Denied inactive", "Quiz", 20, "Midterm", teacher).Success, "deactivated professor cannot create assessments")
                Check(Not revision.ReviewRequestResult(requestId, "Under Review", teacher).Success, "deactivated professor cannot review requests")
            Finally
                DatabaseHelper.ExecuteNonQuery("UPDATE teacher SET status='Active' WHERE teacher_id=@Id", New MySqlParameter("@Id", teacher))
            End Try
            DatabaseHelper.ExecuteNonQuery("UPDATE student SET status='Inactive' WHERE student_id=@Id", New MySqlParameter("@Id", student))
            Try
                Check(Not revision.SubmitRequestResult(New RevisionRequest With {.GradeId = gradeId, .Reason = "Inactive test"}, student).Success, "deactivated student cannot submit requests")
            Finally
                DatabaseHelper.ExecuteNonQuery("UPDATE student SET status='Active' WHERE student_id=@Id", New MySqlParameter("@Id", student))
            End Try
            Dim email = Guid.NewGuid().ToString("N") & "@example.test"
            Dim studentId = "S" & Guid.NewGuid().ToString("N").Substring(0, 8)
            Dim teacherId = "T" & Guid.NewGuid().ToString("N").Substring(0, 8)
            Dim password = " test password "
            Dim hash = SecurityHelper.HashPassword(password)
            Dim before = DashboardService.GetStatistics().Rows(0)
            DatabaseHelper.ExecuteNonQuery("INSERT INTO student(student_id,full_name,email,password_hash,program,year_level,section) VALUES(@Id,'Review Login Student',@Email,@Hash,'BSCS',1,'TEST')", New MySqlParameter("@Id", studentId), New MySqlParameter("@Email", email), New MySqlParameter("@Hash", hash))
            DatabaseHelper.ExecuteNonQuery("INSERT INTO teacher(teacher_id,full_name,email,password_hash,department,employment_type) VALUES(@Id,'Review Login Teacher',@Email,@Hash,'CCS','Full-time')", New MySqlParameter("@Id", teacherId), New MySqlParameter("@Email", email), New MySqlParameter("@Hash", hash))
            Dim auth As New AuthService()
            Check(auth.VerifyLogin(email, password, "Student").UserId = studentId, "shared email authenticates selected student role")
            Check(auth.VerifyLogin(email, password, "Teacher").UserId = teacherId, "shared email authenticates selected teacher role")
            Check(auth.VerifyLogin(email, password.Trim(), "Student") Is Nothing, "password spaces are significant")
            DatabaseHelper.ExecuteNonQuery("UPDATE student SET status='Inactive' WHERE student_id=@Id", New MySqlParameter("@Id", studentId))
            Try
                Check(Not auth.ChangePassword("Student", studentId, password, "Changed password").Success, "deactivated session cannot change its password")
            Finally
                DatabaseHelper.ExecuteNonQuery("UPDATE student SET status='Active' WHERE student_id=@Id", New MySqlParameter("@Id", studentId))
            End Try
            Dim adminId = Convert.ToString(DatabaseHelper.ExecuteScalar("SELECT admin_id FROM admin LIMIT 1"))
            Check(New UserService().ResetPassword("Student", studentId, password, adminId).Success, "admin can reset a password containing spaces")
            Check(auth.VerifyLogin(email, password, "Student").UserId = studentId, "reset password preserves spaces at authentication")
            Check(Not New UserService().ResetPassword("Student", studentId, password, "invalid-admin").Success, "invalid admin cannot reset account password")
            Dim after = DashboardService.GetStatistics().Rows(0)
            Check(CInt(after("students")) = CInt(before("students")) + 1 AndAlso CInt(after("teachers")) = CInt(before("teachers")) + 1, "dashboard statistics reflect new database accounts")
            Dim rosterImporter As New RosterImportService()
            Check(rosterImporter.ImportStudents(classId, teacher, New List(Of String()) From {New String() {studentId, "Review Login Student"}}).Success, "student CSV creates a persistent enrollment")
            Check(service.GetEnrollmentsByClass(classId, teacher).Select("student_id='" & studentId & "'").Length = 1, "imported student can be reloaded by identity")
            Check(Not rosterImporter.ImportStudents(classId, teacher, New List(Of String()) From {New String() {studentId, "Wrong name"}}).Success, "roster import rejects ID and name mismatch")
            Check(rosterImporter.ImportStudents(classId, teacher, New List(Of String()) From {New String() {studentId, "Review Login Student"}}).Success, "reimport preserves existing enrollment")
            DatabaseHelper.ExecuteNonQuery("UPDATE enrollment SET running_grade=NULL WHERE class_id=@Class AND student_id=@Student", New MySqlParameter("@Class", classId), New MySqlParameter("@Student", student))
            Check(Not IsDBNull(service.GetEnrollmentsByClass(classId, teacher).Select("student_id='" & student & "'")(0)("running_grade")), "professor reads calculated grade instead of stale cached grade")
            Dim users As New UserService()
            Check(Not users.ToggleAccountStatus("Student", studentId, "invalid-admin").Success, "unauthorized status toggle is rejected")
            Check(users.ToggleAccountStatus("Student", studentId, adminId).Success, "student badge toggles to inactive")
            Check(auth.VerifyLogin(email, password, "Student") Is Nothing, "inactive student cannot log in after badge toggle")
            Check(users.ToggleAccountStatus("Student", studentId, adminId).Success, "student badge toggles back to active")
            Check(auth.VerifyLogin(email, password, "Student") IsNot Nothing, "reactivated student can log in")
            Check(users.ToggleAccountStatus("Teacher", teacherId, adminId).Success, "professor badge toggles to inactive")
            Check(auth.VerifyLogin(email, password, "Teacher") Is Nothing, "inactive professor cannot log in")
            Check(users.ToggleAccountStatus("Teacher", teacherId, adminId).Success, "professor badge toggles back to active")
            Check(Not users.ToggleTeacherEmploymentType(teacherId, "invalid-admin").Success, "unauthorized employment toggle is rejected")
            Check(users.ToggleTeacherEmploymentType(teacherId, adminId).Success, "employment badge toggles to part-time")
            Check(Convert.ToString(DatabaseHelper.ExecuteScalar("SELECT employment_type FROM teacher WHERE teacher_id=@Id", New MySqlParameter("@Id", teacherId))) = "Part-time", "part-time employment persists")
            Check(users.ToggleTeacherEmploymentType(teacherId, adminId).Success, "employment badge toggles back to full-time")
            Check(Convert.ToString(DatabaseHelper.ExecuteScalar("SELECT employment_type FROM teacher WHERE teacher_id=@Id", New MySqlParameter("@Id", teacherId))) = "Full-time", "full-time employment persists")
            Check(Not users.DeleteUser("Student", studentId, adminId).Success, "student with enrollment cannot be deleted")
            Check(Not users.DeleteUser("Teacher", teacher, adminId).Success, "professor with classes or history cannot be deleted")
            Dim deletableStudent = "D" & Guid.NewGuid().ToString("N").Substring(0, 8)
            Check(users.CreateStudentResult(deletableStudent, password, "Delete fixture", Guid.NewGuid().ToString("N") & "@example.test", "BSCS", 1, "TEST", adminId).Success, "create unlinked student through admin service")
            Check(Not users.DeleteUser("Student", deletableStudent, "invalid-admin").Success, "unauthorized deletion is rejected")
            Check(users.DeleteUser("Student", deletableStudent, adminId).Success, "delete permanently removes unlinked student")
            Check(CInt(DatabaseHelper.ExecuteScalar("SELECT COUNT(*) FROM student WHERE student_id=@Id", New MySqlParameter("@Id", deletableStudent))) = 0, "deleted student record is absent")
            Check(CInt(DatabaseHelper.ExecuteScalar("SELECT COUNT(*) FROM auditlog WHERE entity_type='Student' AND entity_id=@Id AND action='Deleted account'", New MySqlParameter("@Id", deletableStudent))) = 1, "permanent deletion leaves an admin audit entry")
            Check(users.DeleteUser("Teacher", teacherId, adminId).Success, "delete permanently removes unlinked professor")
            Check(CInt(DatabaseHelper.ExecuteScalar("SELECT COUNT(*) FROM teacher WHERE teacher_id=@Id", New MySqlParameter("@Id", teacherId))) = 0, "deleted professor record is absent")
            Check(Not users.DeleteUser("Teacher", teacherId, adminId).Success, "second deletion reports missing account")
            Check(Not users.DeleteUser("Admin", adminId, adminId).Success, "admin deletion is not available from account management")
        End Using
    End Sub
    Sub Check(ok As Boolean, name As String)
        If Not ok Then Throw New Exception("FAIL: " & name)
        Console.WriteLine("PASS: " & name)
    End Sub
End Module
